# Example nginx configuration for discovery.dor.lib.umich.edu
# Add this to your nginx server block

location /opensearch-api/ {
    # Remove /opensearch-api prefix and proxy to OpenSearch
    rewrite ^/opensearch-api/(.*) /$1 break;

    proxy_pass https://opensearch.discovery.dor.lib.umich.edu;
    proxy_set_header Host opensearch.discovery.dor.lib.umich.edu;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;

    # Add Basic Auth header
    proxy_set_header Authorization "Basic YWRtaW46RGlzY092ZXJ5MCEyMzRkYXdn";

    # Handle redirects
    proxy_redirect off;
    proxy_ssl_verify off;
}
